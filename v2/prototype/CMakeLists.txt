cmake_minimum_required(VERSION 3.22)
project(WatchPairNativePrototype VERSION 0.1.0 LANGUAGES CXX)
include(../cmake/Dependencies.cmake)
if(NOT TARGET watchpair_media)
  add_subdirectory(../core ${CMAKE_CURRENT_BINARY_DIR}/core)
endif()
add_library(watchpair_native_core src/torrent_service.cpp src/torrent_service.h)
target_link_libraries(watchpair_native_core PUBLIC watchpair_media LibtorrentRasterbar::torrent-rasterbar)
target_include_directories(watchpair_native_core PUBLIC src)
target_compile_definitions(watchpair_native_core PUBLIC QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
if(MSVC)
  target_compile_options(watchpair_native_core PRIVATE /W4)
else()
  target_compile_options(watchpair_native_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(watchpair-v2-prototype src/main.cpp src/window.cpp src/window.h)
target_link_libraries(watchpair-v2-prototype PRIVATE watchpair_native_core watchpair_video)
enable_testing()
add_executable(native-integration-tests tests/integration.cpp)
target_link_libraries(native-integration-tests PRIVATE watchpair_native_core Qt6::Test)
watchpair_native_runtime(watchpair-v2-prototype)
watchpair_native_runtime(native-integration-tests)
add_test(NAME native-integration COMMAND native-integration-tests)
set_tests_properties(native-integration PROPERTIES TIMEOUT 300)
install(TARGETS watchpair-v2-prototype RUNTIME DESTINATION bin)
