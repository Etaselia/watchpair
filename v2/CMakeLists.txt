cmake_minimum_required(VERSION 3.22)
project(WatchPair VERSION 2.0.0 LANGUAGES CXX)
set(WATCHPAIR_APP_VERSION "2.0.0-alpha.4" CACHE STRING "Application release version")
if(NOT WATCHPAIR_APP_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)(-(alpha|beta|rc)\\.(0|[1-9][0-9]*))?$")
  message(FATAL_ERROR "WATCHPAIR_APP_VERSION must be a semantic version, for example 2.0.0-alpha.1. Quote the complete -D argument in PowerShell.")
endif()
include(cmake/Dependencies.cmake)
find_package(Qt6 6.4 REQUIRED COMPONENTS Sql WebSockets Concurrent)
find_package(OpenSSL REQUIRED)
find_package(CURL 7.85 REQUIRED)
find_package(LibArchive REQUIRED)
set(WATCHPAIR_UPDATE_PUBLIC_KEY "" CACHE STRING "Pinned Ed25519 release public key, 64 lowercase hex characters")
if(NOT WATCHPAIR_UPDATE_PUBLIC_KEY STREQUAL "" AND NOT WATCHPAIR_UPDATE_PUBLIC_KEY MATCHES "^[a-f0-9]+$")
  message(FATAL_ERROR "Update key must be lowercase hexadecimal")
endif()
string(LENGTH "${WATCHPAIR_UPDATE_PUBLIC_KEY}" WATCHPAIR_UPDATE_KEY_LENGTH)
if(NOT WATCHPAIR_UPDATE_KEY_LENGTH EQUAL 0 AND NOT WATCHPAIR_UPDATE_KEY_LENGTH EQUAL 64)
  message(FATAL_ERROR "Ed25519 update key must be exactly 32 bytes")
endif()
get_filename_component(WATCHPAIR_QT_PREFIX "${Qt6_DIR}/../../.." ABSOLUTE)
find_program(WATCHPAIR_QTPATHS_EXECUTABLE NAMES qtpaths qtpaths6 HINTS "${WATCHPAIR_QT_PREFIX}/bin" REQUIRED)
file(WRITE "${CMAKE_BINARY_DIR}/qt-runtime-path.txt" "${WATCHPAIR_QTPATHS_EXECUTABLE}\n")
configure_file(cmake/build-info.json.in build-info.json @ONLY)
enable_testing()
add_subdirectory(core)
add_subdirectory(prototype)
add_subdirectory(desktop)
